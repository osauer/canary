package daemon

import (
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/osauer/ibkr/internal/rpc"
)

// TestGenerateHistoricalV171FileAuthority runs only in an isolated archive of
// the immutable v1.7.1 source. It uses that tag's order-journal and
// purge-ledger-v1 writers to preserve the last source format before old
// releases began discarding mismatched purge ledgers.
func TestGenerateHistoricalV171FileAuthority(t *testing.T) {
	output := os.Getenv("CANARY_UPGRADE_FIXTURE_OUTPUT")
	if output == "" {
		t.Skip("maintainer fixture generation only")
	}
	stateHome := filepath.Join(output, "state")
	if err := os.MkdirAll(stateHome, 0o700); err != nil {
		t.Fatal(err)
	}
	t.Setenv("XDG_STATE_HOME", stateHome)

	base := time.Date(2026, 6, 17, 11, 30, 0, 0, time.UTC)
	orderPath, err := defaultOrderJournalPath()
	if err != nil {
		t.Fatal(err)
	}
	contract := rpc.ContractParams{
		ConID:       265598,
		Symbol:      "AAPL",
		SecType:     "STK",
		Exchange:    "SMART",
		PrimaryExch: "NASDAQ",
		Currency:    "USD",
		Multiplier:  1,
	}
	route := func(event orderJournalEvent) orderJournalEvent {
		event.ClientID = 37
		event.Account = "SYNTHETIC-V171-PAPER"
		event.Endpoint = "127.0.0.1:4002"
		event.Mode = "paper"
		event.ConID = contract.ConID
		event.Symbol = contract.Symbol
		event.SecType = contract.SecType
		event.Exchange = contract.Exchange
		event.PrimaryExch = contract.PrimaryExch
		event.Currency = contract.Currency
		event.Multiplier = contract.Multiplier
		event.OrderType = rpc.OrderTypeLMT
		event.TIF = rpc.OrderTIFDay
		return event
	}
	purgeAttempt := route(orderJournalEvent{
		At:              base.Add(-4 * time.Hour),
		Type:            orderJournalEventSendAttempted,
		OrderRef:        "fixture-v171-purge",
		ReservedOrderID: 1701,
		Source:          purgeExecuteSource,
		PurgeID:         "fixture-v171-purge-set",
		LegID:           "fixture-v171-leg",
		Action:          rpc.OrderActionSell,
		Quantity:        5,
		LimitPrice:      121,
		SendState:       orderSendStateSendAttempted,
	})
	purgeFill := purgeAttempt
	purgeFill.At = base.Add(-3 * time.Hour)
	purgeFill.Type = orderJournalEventStatusUpdated
	purgeFill.Status = "Submitted"
	purgeFill.Filled = 4
	purgeFill.Remaining = 1
	purgeFill.AvgFillPrice = 120
	purgeFill.SendState = orderSendStateBrokerAcknowledged

	restoreAttempt := route(orderJournalEvent{
		At:              base.Add(-2 * time.Hour),
		Type:            orderJournalEventSendAttempted,
		OrderRef:        "fixture-v171-restore",
		ReservedOrderID: 1702,
		Source:          purgeRestoreSource,
		PurgeID:         "fixture-v171-purge-set",
		LegID:           "fixture-v171-leg",
		Action:          rpc.OrderActionBuy,
		Quantity:        4,
		LimitPrice:      116,
		SendState:       orderSendStateSendAttempted,
	})
	restoreFill := restoreAttempt
	restoreFill.At = base.Add(-time.Hour)
	restoreFill.Type = orderJournalEventStatusUpdated
	restoreFill.Status = "Submitted"
	restoreFill.Filled = 1
	restoreFill.Remaining = 3
	restoreFill.AvgFillPrice = 115
	restoreFill.SendState = orderSendStateBrokerAcknowledged

	if err := newOrderJournalStore(orderPath).AppendAll([]orderJournalEvent{
		purgeAttempt, purgeFill, restoreAttempt, restoreFill,
	}); err != nil {
		t.Fatal(err)
	}
	purgePath, err := defaultPurgeLedgerPath()
	if err != nil {
		t.Fatal(err)
	}
	ledger := newPurgeLedgerStore(purgePath, func() time.Time { return base })
	if err := ledger.ApplyOrderFill(purgeFill); err != nil {
		t.Fatal(err)
	}
	if err := ledger.ApplyOrderFill(restoreFill); err != nil {
		t.Fatal(err)
	}
	rows, totals, err := ledger.Snapshot("", "")
	if err != nil {
		t.Fatal(err)
	}
	if len(rows) != 1 ||
		rows[0].Status != purgeLedgerStatusActive ||
		rows[0].RemainingQuantity != 3 ||
		rows[0].RestoredQuantity != 1 ||
		totals.ActiveRows != 1 {
		t.Fatalf("generated active restore state rows=%+v totals=%+v", rows, totals)
	}
	for _, path := range []string{orderPath, purgePath} {
		info, err := os.Stat(path)
		if err != nil {
			t.Fatalf("generated fixture %s: %v", path, err)
		}
		if !info.Mode().IsRegular() || info.Mode().Perm() != 0o600 {
			t.Fatalf("generated fixture %s mode=%v", path, info.Mode())
		}
	}
}
