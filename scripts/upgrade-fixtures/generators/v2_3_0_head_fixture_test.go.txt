package daemon

import (
	"os"
	"testing"

	"github.com/osauer/ibkr/v2/internal/daemon/corestore"
)

// TestGenerateHistoricalV230AuthorityHead uses the v2.3.0 daemon's own
// anti-rollback watermark writer after the corestore fixture is complete.
func TestGenerateHistoricalV230AuthorityHead(t *testing.T) {
	path := os.Getenv("CANARY_UPGRADE_FIXTURE_OUTPUT")
	if path == "" {
		t.Skip("maintainer fixture generation only")
	}
	inspection, err := corestore.Inspect(t.Context(), corestore.InspectOptions{Path: path})
	if err != nil {
		t.Fatal(err)
	}
	if inspection.SchemaVersion != 1 || inspection.Head.HeadGeneration != 1 {
		t.Fatalf("unexpected generated authority: %+v", inspection)
	}
	if err := writeAuthorityWatermark(path+".head", inspection.Head); err != nil {
		t.Fatal(err)
	}
	info, err := os.Stat(path + ".head")
	if err != nil {
		t.Fatal(err)
	}
	if !info.Mode().IsRegular() || info.Mode().Perm() != 0o600 {
		t.Fatalf("generated watermark mode=%v", info.Mode())
	}
}
