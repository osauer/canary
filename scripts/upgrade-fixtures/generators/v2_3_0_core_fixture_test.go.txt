package corestore

import (
	"context"
	"crypto/rand"
	"database/sql"
	"os"
	"path/filepath"
	"testing"
	"time"
)

type fixtureRepeatReader byte

func (r fixtureRepeatReader) Read(p []byte) (int, error) {
	for i := range p {
		p[i] = byte(r)
	}
	return len(p), nil
}

// TestGenerateHistoricalV230CoreAuthority runs only in an isolated archive of
// the immutable v2.3.0 source. It uses that tag's migration and typed state
// writer internals with a fixed clock and deterministic entropy, then leaves a
// checkpointed standalone schema-v1 database.
func TestGenerateHistoricalV230CoreAuthority(t *testing.T) {
	path := os.Getenv("CANARY_UPGRADE_FIXTURE_OUTPUT")
	if path == "" {
		t.Skip("maintainer fixture generation only")
	}
	if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
		t.Fatal(err)
	}
	if err := os.Chmod(filepath.Dir(path), 0o700); err != nil {
		t.Fatal(err)
	}
	file, err := os.OpenFile(path, os.O_CREATE|os.O_EXCL|os.O_RDWR, 0o600)
	if err != nil {
		t.Fatal(err)
	}
	if err := file.Close(); err != nil {
		t.Fatal(err)
	}

	originalReader := rand.Reader
	rand.Reader = fixtureRepeatReader(0x42)
	defer func() { rand.Reader = originalReader }()

	ctx := context.Background()
	db, err := sql.Open("sqlite", sqliteDSN(path, defaultBusyTimeout, false))
	if err != nil {
		t.Fatal(err)
	}
	db.SetMaxOpenConns(1)
	db.SetMaxIdleConns(1)
	if err := db.PingContext(ctx); err != nil {
		t.Fatal(err)
	}
	if err := verifyPragmas(ctx, db, defaultBusyTimeout); err != nil {
		t.Fatal(err)
	}
	createdAt := time.Date(2026, 7, 1, 9, 0, 0, 0, time.UTC)
	if err := migrate(ctx, db, currentMigrationPlan(), createdAt); err != nil {
		t.Fatal(err)
	}

	store := &Store{
		db: db, path: path, busyTimeout: defaultBusyTimeout,
		health: Health{Ready: true},
	}
	document := []byte(`{"fixture":true,"sentinel":"v2.3.0-state-survives-upgrade","source_tag":"v2.3.0"}`)
	writtenAt := createdAt.Add(time.Minute)
	observations := []ObservationInput{
		{
			ScopeKey: "market/contracts", Source: "ibkr.tws.contract_details",
			Kind: "contract_cache.snapshot.v3", ObservedAt: writtenAt.Add(-time.Minute),
			ContentType: "application/json",
			Payload: []byte(`{"fixture":true,"classification":"exact-defective-contract-cache"}`),
			MetadataJSON: []byte(`{"synthetic":true,"source_tag":"v2.3.0"}`),
		},
		{
			ScopeKey: "market/contracts", Source: "ibkr.tws.contract_details",
			Kind: "contract_cache.snapshot.v3.control", ObservedAt: writtenAt.Add(-30 * time.Second),
			ContentType: "application/json",
			Payload: []byte(`{"fixture":true,"classification":"near-control-must-survive"}`),
			MetadataJSON: []byte(`{"synthetic":true,"source_tag":"v2.3.0"}`),
		},
	}
	for _, observation := range observations {
		if err := validateObservation(observation); err != nil {
			t.Fatal(err)
		}
	}
	if err := store.criticalMutation(ctx, func(tx *sql.Tx) error {
		if _, err := compareAndSwapStateTx(ctx, tx, StateDocumentCAS{
			ScopeKey: "fixture/history", Kind: "fixture.state.v1",
			ExpectedRevision: 0, JSON: document,
		}, writtenAt); err != nil {
			return err
		}
		if _, err := appendObservationsTx(ctx, tx, observations, writtenAt); err != nil {
			return err
		}
		_, err := advanceHeadTx(ctx, tx, 0, writtenAt)
		return err
	}); err != nil {
		t.Fatal(err)
	}
	if _, err := store.Checkpoint(ctx); err != nil {
		t.Fatal(err)
	}
	if err := store.Close(); err != nil {
		t.Fatal(err)
	}
	info, err := os.Stat(path)
	if err != nil {
		t.Fatal(err)
	}
	if !info.Mode().IsRegular() || info.Mode().Perm() != 0o600 {
		t.Fatalf("generated authority mode=%v", info.Mode())
	}
}
